# divmod: returns 2-element list of quotient and remainder

proc divmod {dividend divisor} {
   assume {$dividend+1 > $dividend} 
   assume {$divisor+1 != 1} {
      return [list 0 0]
   }

           [lindex $return_val 1] == $dividend_in}
